# --- Dash ------------------------------------------------------------------
# The [fastapi] extra pulls in FastAPI + Starlette + Uvicorn + Pydantic.
# DASH_BACKEND (read by lib/backend.py) flips between the bundled Flask backend
# and the FastAPI one; the Docker image pins flask because gunicorn is WSGI.
#
# Stable, deliberately. This pin binds only the docs site — the package itself
# is far looser (`dash>=4.1` in pyproject.toml), and CI measures every minor in
# that range on each push.
dash[fastapi]==4.4.1  # COMPAT-MATRIX: dash

# --- The component under documentation -------------------------------------
# Installed from the working tree, not from PyPI: the docs site must exercise
# the bundle in dash_emoji_mart/ that this repo is about to ship, otherwise a
# broken build would still render a green docs site.
-e .

# site needs Python >= 3.10. The dash_emoji_mart PACKAGE needs only >= 3.9.
dash_mantine_components==2.7.0
dash-iconify==0.1.2

# markdown2dash 0.1.2 declares `gunicorn>=21.2.0,<22.0.0` — a markdown parser
# pinning a WSGI server, and directly against the CVE-driven gunicorn>=23 floor
# below (CVE-2024-6827 and CVE-2024-1135, both request smuggling). pip cannot
# resolve both, so markdown2dash is NOT listed here; it is installed WITHOUT its
# dependency set at every install site:
#
#     pip install --no-deps markdown2dash==0.1.2
#
# The sites: Dockerfile, .github/workflows/ci.yml, .github/workflows/release.yml
# and the README's contributor quickstart. CI asserts the resulting gunicorn
# version INSIDE the built image — that assertion is what keeps the dodge
# honest. markdown2dash's real runtime dependencies (dash, dash-iconify,
# dash-mantine-components, docutils, jsonpath) are all satisfied by this file
# already; the only one that was not is mistune, listed here so nothing quietly
# relies on a transitive install that --no-deps no longer performs.
mistune>=3.0.0
docutils!=0.21
# markdown2dash==0.1.2        <- installed --no-deps, see above

python-frontmatter
# The [flask] extra is just `flask>=2.0`, already present via Dash's bundled
# backend. 2.3.3 is the floor because this site depends on three things that
# landed between 2.0 and 2.3.3: merge semantics for register_page_metadata
# (2.2.0 — without it the root re-registration below silently blanks the home
# page's prose), the universal prerender that writes canonical/og/title
# server-side (2.2/2.3 — which is why templates/index.html no longer carries a
# canonical shim), and the corrected AI-crawler taxonomy (2.3.2/2.3.3).
#
# 2.3.4 was the previous network floor (`resolve_site_title` — the /llms.txt
# H1 and the llms viewer's brand chip read SITE_BRAND instead of a fallback).
#
# 2.5.1 is the 402-instrumentation floor (network 1.3.x sync). It adds
# `configure_access` / `configure_viewer_identity` — the hooks lib/access.py
# hands the page-tier policy to — and the tiered corpus documents
# (/llms-small.txt, /llms-full.txt) that LLMS_SMALL_TIER / LLMS_FULL_TIER
# govern. Below it the tier wiring in run.py is dead code and nothing says so;
# run.py enforces this same floor at boot for exactly that reason.
dash-improve-my-llms[flask]>=2.5.1
python-dotenv>=1.0
pydantic>=2.0
requests>=2.31

# --- Deployment ------------------------------------------------------------
# gunicorn serves the Flask (WSGI) backend in the Dockerfile. The FastAPI
# backend is ASGI and needs uvicorn instead — already pulled in by dash[fastapi].
#
# >=23 is a security floor, not a preference: 21.x and 22.x are both affected by
# CVE-2024-6827 and CVE-2024-1135 (request smuggling via malformed
# Transfer-Encoding). This file said `>=21.2,<22` until the network pass — a
# ceiling inherited wholesale from markdown2dash's dependency block, which is
# why the fix above had to come first. CI asserts the version inside the image.
gunicorn>=23.0.0
